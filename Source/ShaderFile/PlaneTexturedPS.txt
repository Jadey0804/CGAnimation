struct PS_INPUT
{
    float4 Pos       : SV_POSITION;
    float3 Normal    : NORMAL;
    float2 TexCoords : TEXCOORD;
};

float4 PS(PS_INPUT input) : SV_Target0
{
    float2 uv = input.TexCoords * 10.0;
    float checker = fmod(floor(uv.x) + floor(uv.y), 2.0);

    float3 c0 = float3(0.15, 0.15, 0.15);
    float3 c1 = float3(0.85, 0.85, 0.85);
    float3 albedo = lerp(c0, c1, checker);

    float3 N = normalize(input.Normal);
    float3 L = normalize(float3(0.3, 1.0, 0.2));
    float ndl = saturate(dot(N, L));

    return float4(0.15 * albedo + ndl * albedo, 1.0);
}
