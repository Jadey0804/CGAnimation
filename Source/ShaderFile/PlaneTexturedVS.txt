cbuffer staticMeshBuffer : register(b0)
{
    float4x4 W;
    float4x4 VP;
};

struct VS_INPUT
{
    float4 Pos       : POSITION;
    float3 Normal    : NORMAL;
    float3 Tangent   : TANGENT;
    float2 TexCoords : TEXCOORD;
};

struct PS_INPUT
{
    float4 Pos       : SV_POSITION;
    float2 TexCoords : TEXCOORD0;

    float3 T         : TEXCOORD1;
    float3 B         : TEXCOORD2;
    float3 N         : TEXCOORD3;
};

PS_INPUT VS(VS_INPUT input)
{
    PS_INPUT output;

    // position
    float4 worldPos = mul(input.Pos, W);
    output.Pos = mul(worldPos, VP);

    // UV
    output.TexCoords = input.TexCoords;

    // World-space TBN (够用版本：W 的 3x3 变换 + normalize)
    float3 N = normalize(mul(input.Normal,  (float3x3)W));
    float3 T = normalize(mul(input.Tangent, (float3x3)W));
    float3 B = normalize(cross(N, T));

    output.T = T;
    output.B = B;
    output.N = N;

    return output;
}
