cbuffer GrassBuffer : register(b0)
{
    float4x4 W;
    float4x4 VP;
    float3 cameraRight;
    float padding1;
    float3 cameraUp;
    float padding2;
    float grassWidth;
    float grassHeight;
    float2 padding3;
};

struct VS_INPUT
{
    float4 Pos : POSITION;
    float3 Normal : NORMAL;
    float3 Tangent : TANGENT;
    float2 TexCoords : TEXCOORD;
};

struct PS_INPUT
{
    float4 Pos : SV_POSITION;
    float2 TexCoords : TEXCOORD;
};

PS_INPUT VS(VS_INPUT input)
{
    PS_INPUT output;
    
    // ??????????W????????
    float3 worldCenter = float3(W[0][3], W[1][3], W[2][3]);
    
    // ??????W???
    float scaleX = length(float3(W[0][0], W[1][0], W[2][0]));
    float scaleY = length(float3(W[0][1], W[1][1], W[2][1]));
    
    // Billboard: ????right?up????quad
    // input.Pos.x ? -0.5 ? 0.5?input.Pos.y ? 0 ? 1
    float3 right = normalize(cameraRight);
    float3 up = float3(0, 1, 0);  // ????????
    
    // ?????????
    float3 worldPos = worldCenter;
    worldPos += right * input.Pos.x * scaleX * grassWidth;   // ????
    worldPos += up * input.Pos.y * scaleY * grassHeight;     // ????
    
    // ???????
    output.Pos = mul(float4(worldPos, 1.0), VP);
    output.TexCoords = input.TexCoords;
    
    return output;
}
